extends "res://scripts/game/invocation_magic.gd"
## Light and excavation share the film clock; original wall art stays intact.
const GATE_CENTER := Vector2(40,-350)
var opening := 0.0
var charge := 0.0
var source := Vector2.ZERO
var wall_material: ShaderMaterial

func bind_wall(wall: Sprite2D) -> void:
	wall_material=ShaderMaterial.new()
	wall_material.shader=preload("res://shaders/invocation_passage.gdshader")
	wall_material.set_shader_parameter("origin",wall.position+wall.offset)
	wall_material.set_shader_parameter("extent",Vector2(wall.texture.get_size()))
	wall_material.set_shader_parameter("centre",GATE_CENTER)
	wall.material=wall_material

func present(time: float) -> void:
	seconds=time
	opening=smoothstep(26.0,29.5,time)
	charge=smoothstep(24.5,26.0,time)*(1.0-.78*smoothstep(29.0,31.0,time))
	visible=time>24.5
	material.set_shader_parameter("effect_time",time)
	if wall_material:
		wall_material.set_shader_parameter("opening",opening)
		wall_material.set_shader_parameter("charge",charge)
		wall_material.set_shader_parameter("effect_time",time)
	queue_redraw()

func _draw() -> void:
	ribbons.clear()
	if charge<=0.0: return
	_glow(GATE_CENTER,Vector2(250,280),VIOLET,charge*.24)
	var casting := smoothstep(24.5,25.7,seconds)*(1.0-smoothstep(29.,30.,seconds))
	for strand in 3:
		var points := PackedVector2Array()
		for i in 41:
			var f := i/40.0
			var p := source.lerp(GATE_CENTER,f)
			p.x += sin(f*9.0-seconds*2.2+strand*2.1)*18.0*sin(PI*f)
			p.y += sin(f*6.0+seconds+strand)*7.0*sin(PI*f)
			points.append(p)
		_line(points,casting*.44,2.1)
	_glow(source,Vector2(105,115),VIOLET,casting*.5)
	var breaking := smoothstep(26.,27.,seconds)*(1.-smoothstep(29.,30.5,seconds))
	for i in 24:
		var f := fposmod(seconds*.47+i*.618,1.)
		var side := -1. if i%2==0 else 1.
		var p := GATE_CENTER+Vector2(side*(72*opening+f*28),-75+i*6+f*f*70)
		_glow(p,Vector2(21,17),VIOLET,breaking*sin(PI*f)*.21)
		draw_circle(p,1.3+i%3,Color(.51,.43,.58,breaking*sin(PI*f)*.65),true,-1,true)
